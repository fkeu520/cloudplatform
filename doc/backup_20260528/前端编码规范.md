## 一、通用规范

### 1.1 命名规范

| 类型 | 规范 | 正例 | 反例 |
|------|------|------|------|
| **组件名** | PascalCase，多单词 | `UserProfile.vue` `DataTable.vue` | `userProfile.vue` `data-table.vue` |
| **组合式函数** | use 前缀 | `useUser.ts` `useAuth.ts` | `user.ts` `auth.ts` |
| **页面文件** | 小写，短横线连接 | `user-list.vue` `order-detail.vue` | `UserList.vue` `orderDetail.vue` |
| **普通文件** | 小写，短横线连接 | `utils/format.ts` `api/user.ts` | `utils/formatDate.ts` `api/UserApi.ts` |
| **常量** | 全大写，下划线 | `MAX_PAGE_SIZE` `DEFAULT_AVATAR` | `maxPageSize` `defaultAvatar` |
| **变量/函数** | camelCase | `userName` `getUserList()` | `user_name` `get_user_list()` |
| **类型/接口** | PascalCase | `interface UserInfo` `type StatusType` | `interface userInfo` `type statusType` |
| **样式类名** | BEM 命名 | `.user-card__title--active` | `.userCardTitleActive` `.title` |
| **路由名** | 小写，短横线 | `/user-management/list` | `/userManagement/list` |
| **事件名** | 小写，短横线 | `@click="handle-submit"` | `@click="handleSubmit"` |

## 三、状态管理规范

### 3.1 Pinia Store 规范

```typescript
// ✅ Store 文件结构
// store/modules/user.ts
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import type { UserInfo, LoginParams } from '@/types/user'
import { login as loginApi, getUserInfo } from '@/api/auth'

export const useUserStore = defineStore('user', () => {
  // ============ State ============
  const token = ref<string>('')
  const userInfo = ref<UserInfo | null>(null)
  const permissions = ref<string[]>([])
  
  // ============ Getters ============
  const isLoggedIn = computed(() => !!token.value)
  const isAdmin = computed(() => userInfo.value?.role === 'ADMIN')
  const hasPermission = computed(() => (perm: string) => {
    return permissions.value.includes(perm)
  })
  
  // ============ Actions ============
  async function login(params: LoginParams) {
    const res = await loginApi(params)
    token.value = res.token
    // 持久化到 localStorage
    localStorage.setItem('token', res.token)
    return res
  }
  
  async function fetchUserInfo() {
    const res = await getUserInfo()
    userInfo.value = res
    permissions.value = res.permissions || []
    return res
  }
  
  function logout() {
    token.value = ''
    userInfo.value = null
    permissions.value = []
    localStorage.removeItem('token')
  }
  
  // ============ Return ============
  return {
    token,
    userInfo,
    permissions,
    isLoggedIn,
    isAdmin,
    hasPermission,
    login,
    fetchUserInfo,
    logout
  }
})
```

## 四、API 调用规范

### 4.1 请求封装

```typescript
// utils/request.ts
import axios from 'axios'
import type { AxiosInstance, AxiosRequestConfig, AxiosResponse } from 'axios'
import { useUserStore } from '@/store/modules/user'

// 创建实例
const request: AxiosInstance = axios.create({
  baseURL: import.meta.env.VITE_API_BASE_URL,
  timeout: 10000,
  headers: {
    'Content-Type': 'application/json'
  }
})

// 请求拦截器
request.interceptors.request.use(
  (config) => {
    const token = useUserStore().token
    if (token) {
      config.headers.Authorization = `Bearer ${token}`
    }
    return config
  },
  (error) => {
    return Promise.reject(error)
  }
)

// 响应拦截器
request.interceptors.response.use(
  (response: AxiosResponse) => {
    const { code, data, message } = response.data
    
    if (code === '200' || code === 200) {
      return data
    }
    
    // 业务错误
    return Promise.reject(new Error(message || '请求失败'))
  },
  (error) => {
    // HTTP 错误处理
    if (error.response) {
      const { status } = error.response
      switch (status) {
        case 401:
          useUserStore().logout()
          window.location.href = '/login'
          break
        case 403:
          // 无权限提示
          break
        case 500:
          // 服务器错误提示
          break
      }
    }
    return Promise.reject(error)
  }
)

export default request
```

### 4.2 API 模块组织

```typescript
// api/user.ts
import request from '@/utils/request'
import type { UserInfo, UserQuery, UserPageResult } from '@/types/user'

export const userApi = {
  // 获取用户列表
  getUserList(params: UserQuery) {
    return request.get<UserPageResult>('/api/v1/users', { params })
  },
  
  // 获取用户详情
  getUserById(id: string) {
    return request.get<UserInfo>(`/api/v1/users/${id}`)
  },
  
  // 创建用户
  createUser(data: Partial<UserInfo>) {
    return request.post<UserInfo>('/api/v1/users', data)
  },
  
  // 更新用户
  updateUser(id: string, data: Partial<UserInfo>) {
    return request.put<UserInfo>(`/api/v1/users/${id}`, data)
  },
  
  // 删除用户
  deleteUser(id: string) {
    return request.delete(`/api/v1/users/${id}`)
  }
}
```
